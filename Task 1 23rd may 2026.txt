Table Structure
-- Create Departments Table
CREATE TABLE Departments (
dept_id INT PRIMARY KEY,
dept_name VARCHAR(50) NOT NULL,
location VARCHAR(50)
);

-- Create Jobs Table
CREATE TABLE Jobs (
job_id INT PRIMARY KEY,
job_title VARCHAR(50),
min_salary DECIMAL(10,2),
max_salary DECIMAL(10,2)
);
INSERT INTO Jobs (job_id, job_title, min_salary, max_salary) VALUES
(1, 'Manager', 80000.00, 150000.00),
(2, 'Developer', 60000.00, 110000.00),
(3, 'Analyst', 50000.00, 90000.00),
(4, 'Intern', 30000.00, 45000.00);
-- Create Employees Table
CREATE TABLE Employees (
emp_id INT PRIMARY KEY,
name VARCHAR(50) NOT NULL,
manager_id INT,
dept_id INT,
job_id INT,
salary DECIMAL(10,2),
hire_date DATE,
FOREIGN KEY (dept_id) REFERENCES Departments(dept_id),
FOREIGN KEY (job_id) REFERENCES Jobs(job_id)
);

1.Select e.name, j.job_title, d.dept_name from Employees e join Jobs j on e.job_id = j.job_id join Departments d on e.dept_id = d.dept_id;

3.Select d.dept_name, SUM(e.salary) as total_salary from Employees e join Departments d on e.dept_id = d.dept_id group by d.dept_name;
[GROUP BY] Calculate the total salary expenditure for each department name (not ID).

4.Select d.dept_name from Employees e join Departments d on e.dept_id = d.dept_id group by d.dept_name having COUNT(*) > 2 and 
avg(e.salary) > 70000;
[HAVING] Display departments that have more than 2 employees and an average salary above 70,000.

5.Select e.name as employee, m.name as manager from Employees e left join Employees m on e.manager_id = m.emp_id;
[SELF JOIN] Retrieve a list of employee names and their respective manager's names.

8.Select Year(hire_date) as year,COUNT(*) as total_employees from Employees group by Year(hire_date);
[GROUP BY] Find the number of employees hired each year.

9.Select j.job_title from Employees e join Jobs j on e.job_id = j.job_id group by j.job_title having MAX(e.salary) < 100000;
[HAVING] List job titles where the maximum salary of an employee in that role is less than 100,000.

10.Select name from Employees where dept_id =(
    select dept_id
    from Employees
    where name = 'David'
);
[SUBQUERY] Find the names of employees who work in the same department as 'David'.

11.Select d.dept_name, e.name from Departments d left join Employees e on d.dept_id = e.dept_id;
[MULTI-TABLE JOIN] Show all department names even if they have no employees assigned to them.

12.Select name, salary from Employees where hire_date between '2018-01-01' and '2020-12-31' and salary > 60000;
[COMPLEX FILTER] Find employees who were hired between 2018 and 2020 and earn more than 60,000.

13.Select name, salary from Employees e1 where salary >(
    select avg(salary)
    from Employees e2
    where e1.dept_id = e2.dept_id
);
[CORRELATED SUBQUERY] Find employees who earn more than the average salary of their own department.

14.Select e.name, j.job_title, e.salary from Employees e join Jobs j on e.job_id = j.job_id where e.emp_id = 504;
[JOINS] Find the job title and salary of the employee with ID 504.

15.Select manager_id,MIN(salary) as minimum_salary from Employees where manager_id is not null group by manager_id;
[GROUP BY] For each manager (by ID), find the minimum salary of the people reporting to them.
