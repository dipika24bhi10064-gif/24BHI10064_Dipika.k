-- 1. List the Employee Name, their Department Name, and the Location Name for all employees.
Select e.employee_name, d.dept_name, d.location_id from Employees e join Departments d on e.dept_id = d.dept_id;

-- 2. Retrieve the Project Name, Client Name, and the Total Hours Worked on that project, but only for projects managed by the 'Consulting' department.
Select p.project_name, p.client_name, SUM(a.hours_worked) as total_hours from Projects p 
join Assignments ass on p.project_id = ass.project_id join Departments d on p.dept_id = d.dept_id where d.dept_name = 'Consulting'
group by p.project_name, p.client_name;

-- 3. Find the names of Employees who have worked on a project with Client Name 'GlobalTech Solutions'. List the Employee Name and Project Name.
Select e.employee_name, p.project_name from Employees e join Assignments ass on e.employee_id = ass.employee_id
join Projects p on a.project_id = p.project_id where p.client_name = 'GlobalTech Solutions';

-- 4. List all Departments that have no Employees currently assigned to them.
Select d.dept_name from Departments d left join Employees e on d.dept_id = e.dept_id where e.employee_id is null;

-- 6. Calculate the Average Hours Worked per Employee across all projects. List the Employee Name and the average hours.
Select e.employee_name, avg(ass.hours_worked) as avg_hours from Employees e join Assignments a on e.employee_id = ass.employee_id
group by e.employee_name;

-- 7. Find the Department Name(s) that have an average salary above the company-wide average salary.
Select d.dept_name from Employees e join Departments d on e.dept_id = d.dept_id group by d.dept_name
having avg(e.salary) > (select avg(salary) from Employees);

-- 8. Identify the Client Name(s) whose projects have a total combined hours worked exceeding 500 hours.
Select p.client_name, SUM(ass.hours_worked) as total_hours from Projects p join Assignments ass on p.project_id = ass.project_id
group by p.client_name having SUM(ass.hours_worked) > 500;

-- 10. Calculate the total number of employees assigned to each Project, but only include projects that have more than 5 employees assigned.
Select p.project_name, COUNT(distinct ass.employee_id) as employee_count from Projects p join Assignments ass on p.project_id = ass.project_id
group by p.project_name having COUNT(distinct a.employee_id) > 5;

-- 12. Retrieve the Department Name(s) that have employees assigned to more than one project.
Select d.dept_name from Departments d join Employees e on d.dept_id = e.dept_id join Assignments ass on e.employee_id = ass.employee_id
group by d.dept_name having COUNT(distinct ass.project_id) > 1;

-- 13. Write a query to list the Employee Name and Project Name for any employee who has worked less than 10 hours on a project.
Select e.employee_name, p.project_name from Employees e join Assignments ass on e.employee_id = ass.employee_id
join Projects p on ass.project_id = p.project_id where ass.hours_worked < 10;

-- 14. Find the Project Name(s) that have no assigned employees.
Select p.project_name from Projects p left join Assignments ass ON p.project_id = ass.project_id where ass.assignment_id is null;

-- 16. Assuming the Assignments table has a start_date column, find the Employee Name and their oldest project start date.
Select e.employee_name, MIN(ass.start_date) as oldest_project_date from Employees e join Assignments ass on e.employee_id = ass.employee_id 
group by e.employee_name;

-- 17. Write a MySQL query using the DATE_FORMAT function to list all projects that were active in the year 2023.
Select project_name, DATE_FORMAT(start_date, '%Y') as project_year from Projects where DATE_FORMAT(start_date, '%Y') = '2023';

-- 18. Use the MySQL-specific IFNULL function to list all Employee Names. If an employee has no manager, display "CEO" instead of NULL.
Select employee_name,IFNULL(CAST(manager_id as CHAR), 'CEO') as manager_status from Employees;

-- 22. Write a query using a Common Table Expression (CTE) to find the Project Name and the number of unique employees assigned to it.
Select e.employee_name,m.employee_name as manager_name from Employees e left join Employees m on e.manager_id = m.employee_id;

-- 23. Find the Employee Name(s) who are assigned to the same projects as Employee ID 101.
Select distinct e.employee_name from Employees e join  Assignments ass on e.employee_id = ass.employee_id where ass.project_id in (
    select project_id from Assignments where employee_id = 101) and e.employee_id <> 101;
